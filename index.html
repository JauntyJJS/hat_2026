Clean Obstetrics and Periodontal Therapy Dataset

We process the dataset so that we can use it to create summary tables using gtsummary.

Code 
opt_data <- medicaldata::opt |> 
  dplyr::select(c(
    "PID", "Group",
    "Age", "Education",
    "Black", "White", "Asian", "Nat.Am", "Hisp",
    "BMI", "Diabetes", "Hypertension", "Drug.Add", "Use.Alc", "Use.Tob",
    "Prev.preg", "Spont.ab", "Induced.ab", "Any.stillbirth", 
    "Live.PTB", "N.living.kids",
    "Birth.outcome", "Preg.ended...37.wk", "Birthweight", "GA.at.outcome"
    )
  ) |> 
  dplyr::mutate(
    dplyr::across(dplyr::where(is.factor), stringr::str_trim)
  ) |> 
  dplyr::mutate(
    Group = dplyr::case_when(
      .data[["Group"]] == "C"  ~ "Control",
      .data[["Group"]] == "T"  ~ "Treatment"
    ) |> 
      factor(levels = c("Control", "Treatment")
    ),
    Education = dplyr::case_when(
      .data[["Education"]] == "LT 8 yrs"  ~ "≤8 yr",
      .data[["Education"]] == "8-12 yrs"  ~ "9-12 yr",
      .data[["Education"]] == "MT 12 yrs"  ~ "> 12 yr"
    ) |> 
      factor(levels = c("≤8 yr", "9-12 yr", "> 12 yr")
    ),  
    Hisp = dplyr::case_when(
      .data[["Hisp"]] == ""  ~ "No",
      .default = .data[["Hisp"]]
    ),
    dplyr::across(
      .cols = c(
        "Black", "White", "Asian", "Nat.Am", "Hisp"
      ),
      .fns = \(x){
        factor(x, levels = c("No", "Yes"))
      }
    ),
    Hypertension = dplyr::case_when(
      .data[["Hypertension"]] == "N"  ~ "No",
      .data[["Hypertension"]] == "Y"  ~ "Yes",
    ),
    dplyr::across(
      .cols = c("Diabetes", "Drug.Add", "Use.Alc", "Use.Tob"),
      .fns = \(x){
        dplyr::case_when(
          x == ""  ~ NA,
          .default = x
        )
      }
    ),
    dplyr::across(
      .cols = c(
        "Hypertension", "Diabetes", 
        "Drug.Add", "Use.Alc", "Use.Tob"),
      .fns = \(x){
        factor(x, levels = c("No", "Yes"))
      }
    ),
    dplyr::across(
      .cols = c("Spont.ab", "Induced.ab", "Any.stillbirth",
                "Live.PTB"),
      .fns = \(x){
        dplyr::case_when(
          x == ""  ~ NA,
          .default = x
        )
      }
    ),
    have_living_child = dplyr::case_when(
      is.na(.data[["N.living.kids"]]) ~ "No",
      .data[["N.living.kids"]] == 0 ~ "No",
      .data[["N.living.kids"]] %in% c("1", "2", "3", "4", "5", "6", "7", "8", "9") ~ "Yes",
      .default = "Allocation"
    ),
    previous_live_preterm_birth = dplyr::case_when(
      .data[["Prev.preg"]] == "No" ~ NA,
      .data[["Prev.preg"]] == "Yes" & is.na(.data[["N.living.kids"]]) & is.na(.data[["Live.PTB"]]) ~ NA,
      .data[["Prev.preg"]] == "Yes" & is.na(.data[["N.living.kids"]]) & .data[["Live.PTB"]] == "No" ~ NA,
      .data[["Prev.preg"]] == "Yes" & .data[["N.living.kids"]] %in% c("0") & .data[["Live.PTB"]] == "No" ~ NA,
      .data[["Prev.preg"]] == "Yes" & .data[["N.living.kids"]] %in% c("1", "2", "3", "4", "5", "6", "7", "8", "9") & .data[["Live.PTB"]] == "No" ~ "No",
      .data[["Prev.preg"]] == "Yes" & is.na(.data[["N.living.kids"]]) & .data[["Live.PTB"]] == "Yes" ~ "Yes",
      .data[["Prev.preg"]] == "Yes" & .data[["N.living.kids"]] %in% c("1", "2", "3", "4", "5", "6", "7", "8", "9") & .data[["Live.PTB"]] == "Yes" ~ "Yes",
      .default = "Allocation"
    ),
    Spont.ab = dplyr::case_when(
      .data[["Prev.preg"]] == "No" ~ NA,
      .data[["Prev.preg"]] == "Yes" & .data[["Spont.ab"]] == "No" ~ "No",
      .data[["Prev.preg"]] == "Yes" & is.na(.data[["Spont.ab"]]) ~ "No",
      .data[["Prev.preg"]] == "Yes" & .data[["Spont.ab"]] == "Yes" ~ "Yes",
      .default = "Allocation"
    ),
    Induced.ab = dplyr::case_when(
      .data[["Prev.preg"]] == "No" ~ NA,
      .data[["Prev.preg"]] == "Yes" & .data[["Induced.ab"]] == "No" ~ "No",
      .data[["Prev.preg"]] == "Yes" & is.na(.data[["Induced.ab"]]) ~ "No",
      .data[["Prev.preg"]] == "Yes" & .data[["Induced.ab"]] == "Yes" ~ "Yes",
      .default = "Allocation"
    ),
    Any.stillbirth = dplyr::case_when(
      .data[["Prev.preg"]] == "No" ~ NA,
      .data[["Prev.preg"]] == "Yes" & .data[["Any.stillbirth"]] == "No" ~ "No",
      .data[["Prev.preg"]] == "Yes" & is.na(.data[["Any.stillbirth"]]) ~ "No",
      .data[["Prev.preg"]] == "Yes" & .data[["Any.stillbirth"]] == "Yes" ~ "Yes",
      .default = "Allocation"
    ),
    dplyr::across(
      .cols = c(
        "Prev.preg", "have_living_child", "previous_live_preterm_birth", 
        "Spont.ab", "Induced.ab", "Any.stillbirth"),
      .fns = \(x){
        factor(x, levels = c("No", "Yes"))
      }
    ),
    GA.at.outcome.weeks = .data[["GA.at.outcome"]] / 7,
    Preg_duration_32 = dplyr::case_when(
      .data[["GA.at.outcome.weeks"]] < 32 ~ "Yes",
      .default = "No"
    ),
    Preg_duration_35 = dplyr::case_when(
      .data[["GA.at.outcome.weeks"]] < 35 ~ "Yes",
      .default = "No"
    ),
    Preg_duration_37 = dplyr::case_when(
      .data[["GA.at.outcome.weeks"]] < 37 ~ "Yes",
      .default = "No"
    ),
    Preg_duration_37_regression = dplyr::case_when(
      .data[["GA.at.outcome.weeks"]] < 37 ~ 1,
      .default = 0
    ),
    dplyr::across(
      .cols = c(
        "Preg_duration_32", "Preg_duration_35", 
        "Preg_duration_37"),
      .fns = \(x){
        factor(x, levels = c("No", "Yes"))
      }
    ),
    Birthweight_2500 = dplyr::case_when(
      .data[["Birthweight"]] < 2500 ~ "Yes",
      is.na(.data[["Birthweight"]]) ~ NA,
      .default = "No"
    ),
    Birthweight_2500_regression = dplyr::case_when(
      .data[["Birthweight"]] < 2500 ~ 1,
      is.na(.data[["Birthweight"]]) ~ NA,
      .default = 0
    ),
    Birthweight_1500 = dplyr::case_when(
      .data[["Birthweight"]] < 1500 ~ "Yes",
      is.na(.data[["Birthweight"]]) ~ NA,
      .default = "No"
    ),
    dplyr::across(
      .cols = c(
        "Birthweight_2500", "Birthweight_1500"),
      .fns = \(x){
        factor(x, levels = c("No", "Yes"))
      }
    ),
    GA.at.outcome.weeks = .data[["GA.at.outcome"]] / 7,
    Have_Live_Birth = dplyr::case_when(
      .data[["Birth.outcome"]] == "Live birth" ~ "Yes",
      .default = "No"
    ),
    Preterm_32 = dplyr::case_when(
      .data[["Birth.outcome"]] == "Non-live birth" ~ NA,
      .data[["Birth.outcome"]] == "Live birth"  & .data[["GA.at.outcome.weeks"]] < 32 ~ "Yes",
      .default = "No"
    ),
    Preterm_35 = dplyr::case_when(
      .data[["Birth.outcome"]] == "Non-live birth" ~ NA,
      .data[["Birth.outcome"]] == "Live birth"  & .data[["GA.at.outcome.weeks"]] < 35 ~ "Yes",
      .default = "No"
    ),
    Preterm_37 = dplyr::case_when(
      .data[["Birth.outcome"]] == "Non-live birth" ~ NA,
      .data[["Birth.outcome"]] == "Live birth"  & .data[["GA.at.outcome.weeks"]] < 37 ~ "Yes",
      .default = "No"
    ),
    dplyr::across(
      .cols = c(
        "Have_Live_Birth", "Preterm_32", "Preterm_35", "Preterm_37"),
      .fns = \(x){
        factor(x, levels = c("No", "Yes"))
      }
    )
  )

Age, Race, and Education Summary Table

Below is the code used to create the age, race, and education summary table using gtsummary.

opt_age_race_data <- opt_data |> 
  dplyr::select(
    c("PID", "Group", 
      "Age", "White", "Black", "Hisp", "Education")
  )

opt_age_race_table <- opt_age_race_data |>
  gtsummary::tbl_summary(
    include = c("Age", "White", "Black", "Hisp", "Education"),
    by = "Group",
    label = list(
      "Age" ~ "Age - yr",
      "Hisp" ~ "Hispanic",
      "Education" ~ "Education - no. (%)"
    ),
    statistic = list(
      gtsummary::all_continuous() ~ "{mean}±{sd}",
      gtsummary::all_categorical() ~ "{n} ({p})"
    ),
    digits = list(
      gtsummary::all_continuous() ~ 1,
      gtsummary::all_categorical() ~ c(0, 1)
    )
  ) |> 
  gtsummary::add_variable_group_header(
    header = "Race or ethnic group - no. (%)",
    variables = c("White", "Black", "Hisp")
  ) |> 
  # Need broom R package
  gtsummary::add_p(
    pvalue_fun = gtsummary::label_style_pvalue(digits = 2)
  ) |>
  gtsummary::remove_footnote_header()

Extract ARD from gtsummary

In this section, we exploring the Analysis Ready Data (ARD) from our summary table.

age_race_education_ard_list <- gtsummary::gather_ard(
  opt_age_race_table
)

age_race_education_ard_list |> 
  listviewer::reactjson()

Extract Education Summary ARD

We extract the education summary ARD.

education_summary_ard <- age_race_education_ard_list |> 
  purrr::pluck("tbl_summary") |> 
  cards::apply_fmt_fun() |> 
  cards::unlist_ard_columns(
    -c("stat", "fmt_fun", "warning", "error")
  ) |> 
  dplyr::filter(
    .data[["variable"]] == "Education" &
    .data[["variable_level"]] == "≤8 yr"

Combine and convert as Json

We combine the two ARD together and convert the data frame to json.

education_summary_ard <- age_race_education_ard_list |> 
  purrr::pluck("tbl_summary") |> 
  cards::apply_fmt_fun() |> 
  dplyr::filter(
    .data[["variable"]] == "Education"
  )

education_statistics_ard  <- age_race_education_ard_list |> 
  purrr::pluck("add_p", "Education") |> 
  cards::apply_fmt_fun() 

education_ard <- cards::bind_ard(
  education_summary_ard,
  education_statistics_ard
)

age_race_ard_json <- education_ard |> 
  cards::as_nested_list() |>
  jsonlite::toJSON(
    pretty = TRUE,
    auto_unbox = TRUE, # Convert ["value"] to "value"
    force = TRUE, # Convert NULL to {},
    digits = NA # Full precision
  )

age_race_ard_json |> 
  write(file = "output.json")

ARD without gtsummary

It is also possible to obtain the Analysis Ready Data (ARD) directly from the data frame instead of the gtsummary summary table.

ARD from Education only

Here is how it can be done with the education (categorical) data using cards::ard_tabulate and cardx::ard_stats_chisq_test.

education_summary_ard_from_df <- opt_age_race_data |> 
  cards::ard_tabulate(
    variables = c("Education"),
    by = "Group"
  ) |> 
  cards::update_ard_fmt_fun(
    stat_names = c("n"),
    fmt_fun = gtsummary::label_style_number(digits = 0)
  ) |> 
  cards::update_ard_fmt_fun(
    stat_names = c("p"),
    fmt_fun = gtsummary::label_style_number(digits = 1, scale = 100)
  )

education_statistics_ard_from_df <- opt_age_race_data |> 
  cardx::ard_stats_chisq_test(
    variables = c("Education"),
    by = "Group",
    correct = FALSE
  ) |> 
  cards::update_ard_fmt_fun(
    stat_names = c("statistic", "parameter"),
    fmt_fun = gtsummary::label_style_sigfig(digits = 3)
  ) |> 
  cards::update_ard_fmt_fun(
    stat_names = c("p.value"),
    fmt_fun = gtsummary::label_style_pvalue(digits = 2)
  )

combine_ard <- cards::bind_ard(
  education_summary_ard_from_df,
  education_statistics_ard_from_df
)
combine_ard |> 
  reactable::reactable(
    filterable = TRUE,
    paginationType = "jump",
    defaultPageSize = 3,
    minRows = 3)

ARD from Age, Race and Education

Here is how it can be done.

  • For Education (categorical), use cards::ard_tabulate.
  • For Age (continuous), use cards::ard_summary.
  • For White, Black, Hisp (dichotomous), use cards::ard_tabulate_value.

We then combine using cards::ard_stack.

age_race_education_ard_from_df <- opt_age_race_data |>
  cards::ard_stack(
    cards::ard_summary(
      variables = c("Age"),
      statistic = list(
        Age = cards::continuous_summary_fns(
          summaries = c("mean", "sd")
        )
      ),
      stat_label = list(
        Age = list(mean = "Mean", sd = "SD")
      ),
      fmt_fun = list(
        Age = list(
          mean = \(x) gtsummary::style_number(x, digits = 1)
        )
      )
    ),
    # cardx::ard_stats_wilcox_test(
    #   variables = "Age",
    #   conf.int = TRUE,
    # ),
    cards::ard_tabulate_value(variables = c("White", "Black", "Hisp")),
    cards::ard_tabulate(variables = c("Education")),
    .by = "Group",
    .missing = TRUE,
    .attributes = TRUE,
    .total_n = TRUE
  )
